local function redact(value)
	local text = tostring(value)
	local token = BGS.WEBHOOK_URL:match("/webhooks/%d+/([^?]+)")
	for _, secret in ipairs({BGS.WEBHOOK_URL, token or ""}) do
		if secret ~= "" then
			local pattern = secret:gsub("([^%w])", "%%%1")
			text = text:gsub(pattern, "[redacted]")
		end
	end
	return text
end
local function log(message)
	print("BGS: " .. redact(message))
end
local function warnError(message)
	warn("BGS: " .. redact(message))
end
local function bounded(seconds, callback, ...)
	local arguments = table.pack(...)
	local result
	local thread = task.spawn(function()
		result = table.pack(pcall(callback, table.unpack(arguments, 1, arguments.n)))
	end)
	local deadline = os.clock() + seconds
	while not result and runtime.Running and os.clock() < deadline do
		task.wait(0.1)
	end
	if not result then
		pcall(task.cancel, thread)
		return false, "operation timed out or exporter stopped"
	end
	return table.unpack(result, 1, result.n)
end
local function request(options)
	local ok, response = bounded(20, HttpRequest, options)
	if not ok then
		return nil, redact(response)
	end
	if type(response) ~= "table" then
		return nil, "HTTP function returned no response table"
	end
	return response
end
local function responseStatus(response)
	return response and tonumber(response.StatusCode or response.Status or response.status_code or response.status)
end
local function responseBody(response)
	return response and (response.Body or response.body or response.ResponseBody)
end
local function jsonSafe(value, seen, depth)
	local kind = typeof(value)
	if kind == "nil" or kind == "boolean" or kind == "string" then return value end
	if kind == "number" then
		if value == value and math.abs(value) < math.huge then return value end
		return tostring(value)
	end
	if kind == "Instance" then return value:GetFullName() end
	if kind ~= "table" then return tostring(value) end
	seen, depth = seen or {}, depth or 0
	if seen[value] or depth > 24 then return "[cyclic or deeply nested value]" end
	seen[value] = true
	local output, count, maximum, array = {}, 0, 0, true
	for key in pairs(value) do
		count = count + 1
		if type(key) ~= "number" or key < 1 or key % 1 ~= 0 then
			array = false
		else
			maximum = math.max(maximum, key)
		end
	end
	array = array and maximum == count and count > 0
	for key, item in pairs(value) do
		output[array and key or tostring(key)] = jsonSafe(item, seen, depth + 1)
	end
	seen[value] = nil
	return output
end
local function envelope(kind, data)
	return {
		Type = kind,
		GeneratedAtUTC = os.date("!%Y-%m-%dT%H:%M:%SZ"),
		PlaceId = game.PlaceId,
		JobId = game.JobId,
		Version = BGS_PAYLOAD_VERSION,
		Data = jsonSafe(data),
	}
end
local function uploadJsonAttachment(filename, contents)
	local boundary = "BGS" .. HttpService:GenerateGUID(false):gsub("-", "")
	local payload = HttpService:JSONEncode({
		content = "BGS export: " .. filename,
		allowed_mentions = {parse = {}},
		attachments = {{id = 0, filename = filename}},
	})
	local body = table.concat({
		"--" .. boundary,
		'Content-Disposition: form-data; name="payload_json"',
		"Content-Type: application/json", "", payload,
		"--" .. boundary,
		('Content-Disposition: form-data; name="files[0]"; filename="%s"'):format(filename),
		"Content-Type: application/json", "", contents,
		"--" .. boundary .. "--", "",
	}, "\r\n")
	local url = BGS.WEBHOOK_URL:match("^[^?]+") .. "?wait=true"
	for attempt = 1, 3 do
		if not runtime.Running then return false, "exporter stopped" end
		local response, requestError = request({
			Url = url, Method = "POST",
			Headers = {["Content-Type"] = "multipart/form-data; boundary=" .. boundary},
			Body = body,
		})
		local status = responseStatus(response)
		if status and status >= 200 and status < 300 then
			runtime.Sent = runtime.Sent + 1
			log(("Uploaded %s (%d bytes)"):format(filename, #contents))
			return true
		end
		if status == 401 or status == 403 or status == 404 then
			runtime.Running = false
			return false, ("Webhook rejected the request (HTTP %d). Replace or rotate the webhook in BGS-main.txt."):format(status)
		end
		if status == 429 and attempt < 3 then
			local ok, decoded = pcall(HttpService.JSONDecode, HttpService, responseBody(response) or "")
			local delay = ok and type(decoded) == "table" and tonumber(decoded.retry_after) or 5
			delay = delay or 5
			if delay > 120 then return false, "Webhook rate limit exceeds 120 seconds; retry later." end
			task.wait(math.max(1, delay) + 0.25)
		else
			return false, redact(requestError or ("HTTP %s: %s"):format(tostring(status), tostring(responseBody(response)):sub(1, 400)))
		end
	end
	return false, "Webhook retry limit reached"
end
local function enqueue(kind, data)
	local contents = HttpService:JSONEncode(envelope(kind, data))
	if #contents > 4 * 1024 * 1024 then
		error("BGS: " .. kind .. " exceeds the 4 MiB export budget; export smaller batches.")
	end
	if #runtime.Queue >= 64 then error("BGS: upload queue is full; refusing to discard data silently.") end
	table.insert(runtime.Queue, {Name = kind .. ".json", Contents = contents})
end
local function enqueueItems(kind, items)
	if #items == 0 then enqueue(kind, {Items = {}, Count = 0}) return end
	local pages, page, bytes = {}, {}, 0
	for _,item in ipairs(items) do
		local itemBytes = #HttpService:JSONEncode(jsonSafe(item))
		if #page > 0 and bytes + itemBytes > 3 * 1024 * 1024 then
			table.insert(pages, page)
			page, bytes = {}, 0
		end
		table.insert(page, item)
		bytes = bytes + itemBytes + 1
	end
	table.insert(pages, page)
	for index, itemsPage in ipairs(pages) do
		enqueue(#pages == 1 and kind or kind .. "_" .. index, {Items = itemsPage, Part = index, Parts = #pages, Count = #items})
	end
end
local function stage(name, callback, timeout)
	task.spawn(function()
		runtime.Stages[name] = {State = "running"}
		log("Starting " .. name)
		local ok, message = bounded(timeout or 90, callback)
		runtime.Stages[name] = {State = ok and "complete" or "failed", Error = not ok and redact(message) or nil}
		if not ok then
			warnError(name .. ": " .. redact(message))
			if runtime.Running then
				local queued, queueError = pcall(enqueue, name .. "_error", runtime.Stages[name])
				if not queued then warnError(queueError) end
			end
		else
			log("Finished " .. name)
		end
	end)
end

		else

		end
	end
end)

local function parseOriginalStatKeys(source)
local result = {}
if type(source) ~= "string" then
return result
end
local currentName
local readingBuffs = false
for rawLine in source:gmatch("[^\r\n]+") do
local line = rawLine:gsub("%-%-.*$", "")
if readingBuffs then
if line:match("^%s*}%s*,?%s*$") then
readingBuffs = false
else
local key =
line:match('^%s*%["(.-)"%]%s*=')
or line:match("^%s*([%a_][%w_]*)%s*=")
if key then
table.insert(result[currentName], key)
end
end
else
local quotedName = line:match('^%s*%["(.-)"%]%s*=%s*{%s*,?%s*$')
local plainName = line:match("^%s*([%a_][%w_]*)%s*=%s*{%s*,?%s*$")
if quotedName then
currentName = quotedName
elseif plainName and plainName ~= "Buffs" then
currentName = plainName
end
if currentName and line:match("^%s*Buffs%s*=%s*{%s*,?%s*$") then
result[currentName] = {}
readingBuffs = true
end
end
end
return result
end

local function cleanStableImageUrl(value)
if type(value) ~= "string" then
return nil
end
value = value:gsub("\\u0026", "&")
value = value:match("^%s*(.-)%s*$") or value
if not value:match("^https://tr%.rbxcdn%.com/") then
return nil
end
return value:match("^[^?]+")
end
local function extractAssetId(value)
if type(value) == "number" then
return tostring(math.floor(value))
end
if type(value) ~= "string" then
return nil
end
return value:match("rbxassetid://(%d+)")
or value:match("[?&]id=(%d+)")
or value:match("rbxthumb://.-id=(%d+)")
or value:match("^%s*(%d+)%s*$")
end

local function parseBountyCatalog(source)
local catalog = {}
local activeBounties = {}
if type(source) ~= "string" then
return catalog, activeBounties
end
for block in source:gmatch("{(.-)}") do
local name = block:match("[Nn]ame%s*=%s*\"(.-)\"")
or block:match("[Nn]ame%s*=%s*'(.-)'")
local active = block:match("[Aa]ctive%s*=%s*true")
local chance = tonumber(block:match("[Cc]hance%s*=%s*([%deE%+%-%.]+)"))
if name then
name = name:gsub("\\'", "'"):gsub('\\"', '"')
local entry = {
Name = name,
Active = active ~= nil,
Chance = chance,
}
table.insert(catalog, entry)
if entry.Active and entry.Chance then
table.insert(activeBounties, {
Name = entry.Name,
Chance = entry.Chance,
})
end
end
end
return catalog, activeBounties
end

local resolvedImagesByAssetId = {}
local function fetchThumbnailBatch(batch)
if #batch == 0 then
return
end
local url =
"https://thumbnails.roblox.com/v1/assets?assetIds="
.. table.concat(batch, ",")
.. "&size="
.. THUMBNAIL_SIZE
.. "&format=Png&isCircular=false"
local response = request({
Url = url,
Method = "GET",
Headers = {
["Accept"] = "application/json",
["User-Agent"] = "Mozilla/5.0",
},
})
if responseStatus(response) ~= 200 then
return
end
local body = responseBody(response)
local success, decoded = pcall(HttpService.JSONDecode, HttpService, body or "")
if not success or type(decoded) ~= "table" or type(decoded.data) ~= "table" then
return
end
for _, item in ipairs(decoded.data) do
local assetId = item.targetId and string.format("%.0f", item.targetId)
local stableUrl = cleanStableImageUrl(item.imageUrl)
if assetId and item.state == "Completed" and stableUrl then
resolvedImagesByAssetId[assetId] = stableUrl
end
end
end
local unresolved = {}
if enabled(Settings.resolveImages) then
unresolved = table.clone(assetIds)
log("Resolving catalog images")
end
for attempt = 1, THUMBNAIL_RETRIES do
for startIndex = 1, #unresolved, THUMBNAIL_BATCH_SIZE do
local batch = {}
local endIndex = math.min(startIndex + THUMBNAIL_BATCH_SIZE - 1, #unresolved)
for index = startIndex, endIndex do
table.insert(batch, unresolved[index])
end
fetchThumbnailBatch(batch)
task.wait(0.1)
end
local nextUnresolved = {}
for _, assetId in ipairs(unresolved) do
if not resolvedImagesByAssetId[assetId] then
table.insert(nextUnresolved, assetId)
end
end
unresolved = nextUnresolved
if #unresolved == 0 then
break
end
if attempt < THUMBNAIL_RETRIES then
task.wait(1.5)
end
end
local function tryLegacyThumbnail(assetId)
local url =
"https://www.roblox.com/asset-thumbnail/image?assetId="
.. assetId
.. "&width=420&height=420&format=png"
local methods = { "HEAD", "GET" }
for _, method in ipairs(methods) do
local response = request({
Url = url,
Method = method,
Redirect = false,
FollowRedirects = false,
Headers = {
["User-Agent"] = "Mozilla/5.0",
},
})
if response then
local candidates = {
response.Url,
response.URL,
response.ResponseUrl,
response.ResponseURL,
response.FinalUrl,
response.FinalURL,
getHeader(response.Headers or response.headers, "location"),
}
for _, candidate in ipairs(candidates) do
local stableUrl = cleanStableImageUrl(candidate)
if stableUrl then
return stableUrl
end
end
local body = responseBody(response)
if type(body) == "string" then
local found = body:match("https://tr%.rbxcdn%.com/[^\"'%s<>]+")
local stableUrl = cleanStableImageUrl(found)
if stableUrl then
return stableUrl
end
end
end
end
return nil
end
if #unresolved > 0 and enabled(Settings.legacyImageFallback) then
print(("Stable thumbnail API left %d unresolved images; trying decal redirect fallback..."):format(#unresolved))
for index, assetId in ipairs(unresolved) do
local stableUrl = tryLegacyThumbnail(assetId)
if stableUrl then
resolvedImagesByAssetId[assetId] = stableUrl
end
if index % 100 == 0 then
print(("Checked %d/%d unresolved image assets..."):format(index, #unresolved))
end
task.wait(0.03)
end
end

local function collectBounties()
	local modules = requiredChild(requiredChild(ReplicatedStorage, "Assets"), "Modules")
	local service = safeRequire(requiredChild(modules, "SecretBountyService"))
	assert(type(service) == "table" and type(service.Get) == "function", "SecretBountyService.Get is unavailable.")
	local schedule, failures, now = {}, {}, os.time()
	for offset = 0, BOUNTY_DAYS - 1 do
		local timestamp = now + offset * 86400
		local ok, bounty = bounded(5, service.Get, service, timestamp)
		if ok and type(bounty) == "table" and type(bounty.Name) == "string" and type(bounty.Egg) == "string"
			and type(bounty.Chance) == "number" and bounty.Chance > 0 and bounty.Chance < math.huge then
			table.insert(schedule, {
				Date = os.date("!%Y-%m-%d", timestamp),
				Pet = bounty.Name, Egg = bounty.Egg,
				Chance = bounty.Chance, Odds = math.ceil(100 / bounty.Chance),
			})
		else
			table.insert(failures, {Date = os.date("!%Y-%m-%d", timestamp), Error = ok and "Invalid bounty result" or redact(bounty)})
		end
	end
	enqueue("bounties", {Source = "SecretBountyService.Get", Schedule = schedule, Errors = failures, Complete = #schedule == BOUNTY_DAYS})
end
local function visible(node)
	while node do
		if node:IsA("GuiObject") and not node.Visible then return false end
		if node:IsA("LayerCollector") and not node.Enabled then return false end
		node = node.Parent
	end
	return true
end
local function textSnapshot(root, onlyVisible)
	local rows = {}
	if root then
		for _, node in ipairs(root:GetDescendants()) do
			if (node:IsA("TextLabel") or node:IsA("TextButton")) and node.Text ~= "" and (not onlyVisible or visible(node)) then
				table.insert(rows, {Path = node:GetFullName(), Name = node.Name, Text = node.Text})
			end
		end
	end
	table.sort(rows, function(a, b) return a.Path < b.Path end)
	return rows
end
local function boardRows(root)
	local rows = {}
	if not root then return rows end
	for _, frame in ipairs(root:GetDescendants()) do
		local name = frame:FindFirstChild("NameBacking")
		local amount = frame:FindFirstChild("AmountBacking")
		local rank = frame:FindFirstChild("RankBacking")
		name = name and name:FindFirstChild("PlayerName")
		amount = amount and amount:FindFirstChild("Amount")
		rank = rank and rank:FindFirstChild("Rank")
		if name and amount and rank and name:IsA("TextLabel") and amount:IsA("TextLabel") and rank:IsA("TextLabel") then
			local value = amount.Text:gsub(",", "")
			table.insert(rows, {
				Path = frame:GetFullName(),
				PlayerName = name.Text,
				Rank = tonumber(rank.Text:match("%d+")),
				RankText = rank.Text,
				Score = value:match("^%d+$") and value or nil,
				ScoreText = amount.Text,
			})
		end
	end
	table.sort(rows, function(a, b)
		if a.Rank ~= b.Rank then return (a.Rank or math.huge) < (b.Rank or math.huge) end
		return a.Path < b.Path
	end)
	return rows
end
local function collectLeaderboards()
	local boards = {}
	for _, name in ipairs({"BubbleBoard", "HatchBoard", "RankBoard"}) do
		local root = workspace:FindFirstChild(name)
		local rows = boardRows(root)
		table.insert(boards, {Name = name, Rows = rows, Available = #rows > 0, Reason = #rows == 0 and "No replicated board rows available." or nil})
	end
	enqueue("leaderboards", {Scope = "Currently replicated public board rows, not hidden or unloaded periods.", Boards = boards})
end
local function collectCompetitiveLeaderboards()
	local root = workspace:FindFirstChild("Competitive")
	local boards = root and root:FindFirstChild("Boards")
	local result = {}
	for _, name in ipairs({"BubblerBoard", "EggBoard"}) do
		local board = boards and boards:FindFirstChild(name)
		local rows = boardRows(board)
		table.insert(result, {Name = name, Rows = rows, Available = #rows > 0, Reason = #rows == 0 and "No replicated competitive rows available." or nil})
	end
	enqueue("competitive_leaderboards", {Scope = "Currently replicated public board rows.", Boards = result})
end

local function snapshotStages()

	if CompetitiveLeaderboardEnabled then stage("competitive_leaderboards", collectCompetitiveLeaderboards) end
	if ExistsEnabled then stage("pet_exists", collectVisibleExists) end
	if TradeScanEnabled then stage("trades", collectVisibleTrade) end
	if ScanningEnabled then
		stage("inventory", collectVisibleInventory)
		stage("player_profiles", function() enqueue("player_profiles", collectPlayerProfiles()) end)
	end
end
