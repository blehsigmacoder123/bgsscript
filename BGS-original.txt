local function redact(value)
	local text = tostring(value)
	local token = BGS.WEBHOOK_URL:match("/webhooks/%d+/([^?]+)")
	for _, secret in ipairs({BGS.WEBHOOK_URL, token or ""}) do
		if secret ~= "" then
			local pattern = secret:gsub("([^%w])", "%%%1")
			text = text:gsub(pattern, "[redacted]")
		end
	end
	return text
end
local function log(message)
	print("BGS: " .. redact(message))

	warn("BGS: " .. redact(message))
end
local function bounded(seconds, callback, ...)
	local arguments = table.pack(...)
	local result
	local thread = task.spawn(function()
		result = table.pack(pcall(callback, table.unpack(arguments, 1, arguments.n)))
	end)
	local deadline = os.clock() + seconds
	while not result and runtime.Running and os.clock() < deadline do
		task.wait(0.1)
	end
	if not result then
		pcall(task.cancel, thread)
		return false, "operation timed out or exporter stopped"
	end
	return table.unpack(result, 1, result.n)
end
local function request(options)
	local ok, response = bounded(20, HttpRequest, options)
	if not ok then
		return nil, redact(response)
	end
	if type(response) ~= "table" then
		return nil, "HTTP function returned no response table"
	end
	return response
end
local function responseStatus(response)
	return response and tonumber(response.StatusCode or response.Status or response.status_code or response.status)
end
local function responseBody(response)
	return response and (response.Body or response.body or response.ResponseBody)
end

local function fetchRankedServers(url)
	local response = request({
		Url = url,
		Method = "GET",
		Headers = {["Accept"] = "application/json"},
	})
	if responseStatus(response) ~= 200 then
		return nil, "server list request returned HTTP " .. tostring(responseStatus(response))
	end

	if not ok or type(decoded) ~= "table" or type(decoded.data) ~= "table" then
		return nil, "server list response was invalid"

end

end

local function teleportToServer(placeId, server)
	if serverHopInProgress or not server or not LocalPlayer then return false end
	serverHopInProgress = true
	local ok, err = pcall(function()
		TeleportService:TeleportToPlaceInstance(placeId, server.id, LocalPlayer)
	end)

end
local function routeTradeHangout()
	if game.PlaceId ~= BGS_PLACE_ID and game.PlaceId ~= TRADE_HANGOUT_PLACE_ID then return end
	routingActive = true
	local function goToTradeHangout()
		local remote = ReplicatedStorage:FindFirstChild("NetworkRemoteFunction")
		if not remote or not remote:IsA("RemoteFunction") then
			return false, "Trade Hangout remote is unavailable"
		end
		local ok, result = bounded(15, remote.InvokeServer, remote, "TradeTeleport")
		if not ok then return false, tostring(result) end
		return result == true, result == true and nil or "Trade Hangout teleport was rejected"
	end
	local function checkTarget()
		local servers, err = fetchRankedServers(TRADE_HANGOUT_API)
		if not servers then
			warnError("Trade hangout server check failed: " .. tostring(err))
			return false
		end
		local target = rankedOpenServer(servers)
		if target and game.PlaceId == TRADE_HANGOUT_PLACE_ID and target.id == game.JobId then
			tradeHangoutReady = true
			routingActive = false
			return true
		end
		if target then
			teleportToServer(TRADE_HANGOUT_PLACE_ID, target)
			return true
		end
		return false
	end
	task.spawn(function()
		local nextBgsRotation = os.clock() + BGS_ROTATION_INTERVAL
		while runtime.Running and not tradeHangoutReady do
			if game.PlaceId == BGS_PLACE_ID then
				local ok, err = goToTradeHangout()
				if not ok then warnError("Go to Trade Hangout failed: " .. tostring(err)) end
				task.wait(SERVER_CHECK_INTERVAL)
			elseif checkTarget() then
				task.wait(1)
			elseif os.clock() >= nextBgsRotation then
				local servers, err = fetchRankedServers(TRADE_HANGOUT_API)
				if servers then
					teleportToServer(TRADE_HANGOUT_PLACE_ID, firstOpenServer(servers, game.JobId))
				else
					warnError("Trade Hangout rotation check failed: " .. tostring(err))
				end
				nextBgsRotation = os.clock() + BGS_ROTATION_INTERVAL
			else
				task.wait(SERVER_CHECK_INTERVAL)
			end
		end
	end)
end
local function jsonSafe(value, seen, depth)
	local kind = typeof(value)
	if kind == "nil" or kind == "boolean" or kind == "string" then return value end
	if kind == "number" then
		if value == value and math.abs(value) < math.huge then return value end
		return tostring(value)
	end
	if kind == "Instance" then return value:GetFullName() end
	if kind ~= "table" then return tostring(value) end
	seen, depth = seen or {}, depth or 0
	if seen[value] or depth > 24 then return "[cyclic or deeply nested value]" end
	seen[value] = true
	local output, count, maximum, array = {}, 0, 0, true
	for key in pairs(value) do
		count = count + 1
		if type(key) ~= "number" or key < 1 or key % 1 ~= 0 then
			array = false
		else
			maximum = math.max(maximum, key)
		end
	end
	array = array and maximum == count and count > 0
	for key, item in pairs(value) do
		output[array and key or tostring(key)] = jsonSafe(item, seen, depth + 1)
	end
	seen[value] = nil
	return output
end

local function uploadJsonAttachment(filename, contents)
	local boundary = "BGS" .. HttpService:GenerateGUID(false):gsub("-", "")
	local payload = HttpService:JSONEncode({
		content = "BGS export: " .. filename,
		allowed_mentions = {parse = {}},
		attachments = {{id = 0, filename = filename}},
	})
	local body = table.concat({
		"--" .. boundary,
		'Content-Disposition: form-data; name="payload_json"',
		"Content-Type: application/json", "", payload,
		"--" .. boundary,
		('Content-Disposition: form-data; name="files[0]"; filename="%s"'):format(filename),
		"Content-Type: application/json", "", contents,
		"--" .. boundary .. "--", "",
	}, "\r\n")
	local url = BGS.WEBHOOK_URL:match("^[^?]+") .. "?wait=true"
	for attempt = 1, 3 do
		if not runtime.Running then return false, "exporter stopped" end
		local response, requestError = request({
			Url = url, Method = "POST",
			Headers = {["Content-Type"] = "multipart/form-data; boundary=" .. boundary},
			Body = body,
		})
		local status = responseStatus(response)
		if status and status >= 200 and status < 300 then
			runtime.Sent = runtime.Sent + 1
			log(("Uploaded %s (%d bytes)"):format(filename, #contents))
			return true
		end
		if status == 401 or status == 403 or status == 404 then
			runtime.Running = false
			return false, ("Webhook rejected the request (HTTP %d). Replace or rotate the webhook in BGS-main.txt."):format(status)
		end
		if status == 429 and attempt < 3 then
			local ok, decoded = pcall(HttpService.JSONDecode, HttpService, responseBody(response) or "")
			local delay = ok and type(decoded) == "table" and tonumber(decoded.retry_after) or 5
			delay = delay or 5
			if delay > 120 then return false, "Webhook rate limit exceeds 120 seconds; retry later." end
			task.wait(math.max(1, delay) + 0.25)
		else
			return false, redact(requestError or ("HTTP %s: %s"):format(tostring(status), tostring(responseBody(response)):sub(1, 400)))
		end
	end
	return false, "Webhook retry limit reached"
end
local function enqueueContents(kind, contents, priority)
	if #contents > 4 * 1024 * 1024 then
		error("BGS: " .. kind .. " exceeds the 4 MiB export budget; export smaller batches.")
	end
	local limit = priority and 96 or 64
	if #runtime.Queue >= limit then error("BGS: upload queue is full; refusing to discard data silently.") end
	local item = {Name = kind .. ".json", Contents = contents, Priority = priority == true}
	if priority then
		local index = 1
		while runtime.Queue[index] and runtime.Queue[index].Priority do index = index + 1 end
		table.insert(runtime.Queue, index, item)
	else
		table.insert(runtime.Queue, item)
	end
end
local function enqueue(kind, data, priority)
	enqueueContents(kind, HttpService:JSONEncode(jsonSafe(data)), priority)
end
local function enqueueItems(kind, items)
	if #items == 0 then enqueue(kind, {}) return end
	local pages, page, bytes = {}, {}, 0
	for _,item in ipairs(items) do
		local itemBytes = #HttpService:JSONEncode(jsonSafe(item))
		if #page > 0 and bytes + itemBytes > 3 * 1024 * 1024 then
			table.insert(pages, page)
			page, bytes = {}, 0
		end
		table.insert(page, item)
		bytes = bytes + itemBytes + 1
	end
	table.insert(pages, page)
	for index, itemsPage in ipairs(pages) do
		enqueue(#pages == 1 and kind or kind .. "_" .. index, itemsPage)
	end
end
local function stage(name, callback, timeout)
	if runtime.Stages[name] and runtime.Stages[name].State == "running" then return end
	task.spawn(function()
		runtime.Stages[name] = {State = "running"}
		log("Starting " .. name)
		local ok, message = bounded(timeout or 90, callback)
		runtime.Stages[name] = {State = ok and "complete" or "failed", Error = not ok and redact(message) or nil}
		if not ok then
			warnError(name .. ": " .. redact(message))
			if runtime.Running then
				local queued, queueError = pcall(enqueue, name .. "_error", runtime.Stages[name])
				if not queued then warnError(queueError) end
			end
		else
			log("Finished " .. name)
		end
	end)
end
log("Payload v" .. BGS_PAYLOAD_VERSION .. " (" .. BGS_PAYLOAD_REVISION .. ") loaded; checking webhook delivery")

	while runtime.Running do
		local item = table.remove(runtime.Queue, 1)

			local ok, err = uploadJsonAttachment(item.Name, item.Contents)
			if not ok then warnError(item.Name .. ": " .. tostring(err)) end
			task.wait(3)
		else
			task.wait(0.25)
		end
	end
end)

local function parseOriginalStatKeys(source)
local result = {}
if type(source) ~= "string" then
return result
end
local currentName
local readingBuffs = false
for rawLine in source:gmatch("[^\r\n]+") do
local line = rawLine:gsub("%-%-.*$", "")
if readingBuffs then
if line:match("^%s*}%s*,?%s*$") then
readingBuffs = false
else
local key =
line:match('^%s*%["(.-)"%]%s*=')
or line:match("^%s*([%a_][%w_]*)%s*=")
if key then
table.insert(result[currentName], key)
end
end
else
local quotedName = line:match('^%s*%["(.-)"%]%s*=%s*{%s*,?%s*$')
local plainName = line:match("^%s*([%a_][%w_]*)%s*=%s*{%s*,?%s*$")
if quotedName then
currentName = quotedName
elseif plainName and plainName ~= "Buffs" then
currentName = plainName
end
if currentName and line:match("^%s*Buffs%s*=%s*{%s*,?%s*$") then
result[currentName] = {}
readingBuffs = true
end
end
end
return result
end

end

end

"https://thumbnails.roblox.com/v1/assets?assetIds="
.. table.concat(batch, ",")
.. "&size="
.. THUMBNAIL_SIZE
.. "&format=Png&isCircular=false"

Url = url,
Method = "GET",
Headers = {
["Accept"] = "application/json",
["User-Agent"] = "Mozilla/5.0",
},
})
if responseStatus(response) ~= 200 then

end

end
local function tryLegacyThumbnail(assetId)
local url =
"https://www.roblox.com/asset-thumbnail/image?assetId="
.. assetId
.. "&width=420&height=420&format=png"
local methods = { "HEAD", "GET" }
for _, method in ipairs(methods) do
local response = request({
Url = url,
Method = method,
Redirect = false,
FollowRedirects = false,
Headers = {
["User-Agent"] = "Mozilla/5.0",
},
})
if response then
local candidates = {
response.Url,
response.URL,
response.ResponseUrl,
response.ResponseURL,
response.FinalUrl,
response.FinalURL,
getHeader(response.Headers or response.headers, "location"),
}
for _, candidate in ipairs(candidates) do
local stableUrl = cleanStableImageUrl(candidate)
if stableUrl then
return stableUrl
end
end
local body = responseBody(response)
if type(body) == "string" then
local found = body:match("https://tr%.rbxcdn%.com/[^\"'%s<>]+")
local stableUrl = cleanStableImageUrl(found)
if stableUrl then
return stableUrl
end
end
end
end
return nil
end
if #unresolved > 0 and enabled(Settings.legacyImageFallback) then
print(("Stable thumbnail API left %d unresolved images; trying decal redirect fallback..."):format(#unresolved))
for index, assetId in ipairs(unresolved) do

if stableUrl then
resolvedImagesByAssetId[assetId] = stableUrl
end
if index % 100 == 0 then
print(("Checked %d/%d unresolved image assets..."):format(index, #unresolved))
end
task.wait(0.03)
end
end

local function bountyPermutation(seed, count, salt)
	local order = {}
	for i = 1, count do order[i] = i end
	local random = Random.new(seed * 131 + salt)
	for i = count, 2, -1 do
		local j = random:NextInteger(1, i)
		order[i], order[j] = order[j], order[i]
	end
	return order
end
local function bountyRotation(seed, count, salt)
	local cycle = math.floor(seed / count)
	local order = bountyPermutation(cycle, count, salt)
	if count > 2 then
		local previous = bountyPermutation(cycle - 1, count, salt)
		if order[1] == previous[count] then order[1], order[2] = order[2], order[1] end
	end
	return order[seed % count + 1]
end
local function predictBounty(timestamp, eggs)
	local seed = math.floor(timestamp / 86400) - 1
	local pet = BountyModel.Pets[bountyRotation(seed, #BountyModel.Pets, 17)]
	local total = 0
	for _, name in ipairs(eggs) do
		total = total + (BountyModel.FavoredEggs[name] and BountyModel.FavoredEggWeight or 1)
	end
	local index = #eggs
	if total == #eggs then
		index = bountyRotation(seed, #eggs, 53)
	else
		local roll = Random.new(seed * 131 + 53):NextNumber(0, total)
		for i, name in ipairs(eggs) do
			roll = roll - (BountyModel.FavoredEggs[name] and BountyModel.FavoredEggWeight or 1)
			if roll < 0 then index = i break end
		end
	end
	return {Name = pet.Name, Chance = pet.Chance, Egg = eggs[index]}
end
local function collectBounties()
	local assets = ReplicatedStorage:FindFirstChild("Assets")
	local modules = assets and assets:FindFirstChild("Modules")
	local bountyScript = modules and modules:FindFirstChild("SecretBountyService")
	local service
	if bountyScript then
		local ok, value = bounded(10, require, bountyScript)
		if ok and type(value) == "table" and type(value.Get) == "function" then
			service = value
		else
			log("Live bounty service unavailable; exporting predictions from the inspected model. Reason: " .. redact(value))
		end
	end
	local schedule = {}
	local today = math.floor(os.time() / 86400) * 86400
	if service then
		for offset = 0, BOUNTY_DAYS do
			local timestamp = today + offset * 86400
			local ok, value = bounded(5, service.Get, service, timestamp)
			if ok and type(value) == "table" and type(value.Name) == "string" and type(value.Egg) == "string"
				and type(value.Chance) == "number" and value.Chance > 0 and value.Chance < math.huge then
				schedule[offset] = value
			else
				service = nil
				log("Live bounty query unavailable; exporting the complete model forecast instead. Reason: " .. (ok and "invalid bounty response" or redact(value)))
				break
			end
		end
	end
	if not service then
		for offset = 0, BOUNTY_DAYS do
			schedule[offset] = predictBounty(today + offset * 86400, BountyModel.Eggs)
		end
	end
	local entries = {}
	for offset = 0, BOUNTY_DAYS do
		local date = os.date("!%Y-%m-%d", today + offset * 86400)
		local bounty = schedule[offset]
		local fields = {
			'"petName": ' .. HttpService:JSONEncode(bounty.Name),
			'"odds": ' .. HttpService:JSONEncode("1 in " .. string.format("%.0f", math.ceil(100 / bounty.Chance))),
			'"chancePercent": ' .. HttpService:JSONEncode(bounty.Chance),
			'"egg": ' .. HttpService:JSONEncode(bounty.Egg),
		}
		table.insert(entries, "  " .. HttpService:JSONEncode(date) .. ": {\n    " .. table.concat(fields, ",\n    ") .. "\n  }")
	end
	enqueueContents("bounties", "{\n" .. table.concat(entries, ",\n") .. "\n}")
	if not service then
		log("Bounties were generated from the inspected " .. BountyModel.Date .. " rules because the live bounty service was unavailable.")
	end
end

	if boardContext then return boardContext end
	boardContext = {placeId = game.PlaceId, gameId = game.GameId}
	local assets = ReplicatedStorage:FindFirstChild("Assets")

local function boardIdentity(node)
	local period, adornee, panel, surface
	for _ = 1, 24 do
		if not node then break end
		if node:IsA("GuiObject") and (node.Name == "AllTime" or node.Name == "Weekly" or node.Name == "Daily") then
			period, panel = node.Name, node
		end
		if BoardNames[node.Name] then return BoardNames[node.Name], period or "current", panel or surface end
		if node:IsA("SurfaceGui") then surface, adornee = node, node.Adornee end
		node = node.Parent
	end
	node = adornee
	for _ = 1, 12 do
		if not node then break end
		if BoardNames[node.Name] then return BoardNames[node.Name], period or "current", panel or surface end
		node = node.Parent
	end
end
local function textValue(root, name)
	local node = root:FindFirstChild(name, true)
	if node and (node:IsA("TextLabel") or node:IsA("TextButton") or node:IsA("TextBox")) then
		local text = node.Text
		if node.RichText then text = text:gsub("<[^>]*>", "") end
		return text:match("^%s*(.-)%s*$")
	end
end
local function boardEnabled(board)
	local competitive = board:sub(1, 12) == "competitive_"
	return competitive and CompetitiveLeaderboardEnabled or not competitive and LeaderboardsEnabled
end
local function rememberBoardPanel(node, changed)
	if not node:IsA("GuiObject") and not node:IsA("SurfaceGui") then return end
	local board, period, panel = boardIdentity(node)
	if not board or not panel or not boardEnabled(board) then return end
	if board:sub(1, 12) ~= "competitive_" and period == "current" then return end
	local key = board .. "_" .. period:lower()
	local known = boardPanels[key]
	if not known or known.panel ~= panel then
		boardPanels[key] = {panel = panel, sourcePath = panel:GetFullName(), lastAttachedAt = os.time()}
		boardRevision = boardRevision + 1
	elseif changed then
		boardRevision = boardRevision + 1
	end
end
local function discoverBoardPanels()
	table.clear(boardModels)
	local roots = {workspace}
	local gui = LocalPlayer:FindFirstChild("PlayerGui")
	if gui then table.insert(roots, gui) end
	for _, root in ipairs(roots) do
		for index, node in ipairs(root:GetDescendants()) do
			if BoardNames[node.Name] then boardModels[BoardNames[node.Name]] = node:GetFullName() end
			if node:IsA("SurfaceGui") or node.Name == "AllTime" or node.Name == "Weekly" or node.Name == "Daily" then
				rememberBoardPanel(node)
			end
			if index % 1000 == 0 then task.wait(0.03) end
		end
	end
end
local function collectBoards()
	boardLastPoll = os.clock()
	discoverBoardPanels()
	local availability, missing = {}, {}
	local gui = LocalPlayer:FindFirstChild("PlayerGui")
	for key, info in pairs(boardPanels) do
		local panel = info.panel
		local attached = panel:IsDescendantOf(workspace) or gui ~= nil and panel:IsDescendantOf(gui)
		if attached then info.lastAttachedAt = os.time() end
		local rows, seen, invalid = {}, {}, 0
		for _, frame in ipairs(panel:GetDescendants()) do
			if frame:IsA("GuiObject") and frame:FindFirstChild("NameBacking") then
				local name, amount, rankText = textValue(frame, "PlayerName"), textValue(frame, "Amount"), textValue(frame, "Rank")
				local rank = rankText and tonumber(rankText:match("^#?%s*(%d+)")) or tonumber(frame.Name)
				local score = amount and amount:gsub("[,%s]", "")
				if name and name:match("%S") and rank and rank > 0 and rank % 1 == 0 and amount and amount:match("%S") then
					local rowKey = tostring(rank) .. ":" .. name
					if not seen[rowKey] then
						seen[rowKey] = true
						local exact = score:match("^%d+$") ~= nil
						local row = {username = name, rank = rank, score = exact and score or nil, scoreText = amount, scoreExact = exact}
						local id = frame:GetAttribute("UserId")
						if type(id) == "number" then row.userid = id end
						table.insert(rows, row)
					end
				else
					invalid = invalid + 1
				end
			end
		end
		table.sort(rows, function(a, b)
			if a.rank ~= b.rank then return a.rank < b.rank end
			return a.username < b.username
		end)
		availability[key] = {
			rows = #rows, invalidRows = invalid, attached = attached == true,
			sourcePath = info.sourcePath,
			lastAttachedAtUTC = os.date("!%Y-%m-%dT%H:%M:%SZ", info.lastAttachedAt),
			source = attached and "replicated_ui" or "previously_observed_panel",
			reason = #rows == 0 and (invalid > 0 and "unreadable_rows" or "rows_not_loaded") or nil,
		}
		if #rows > 0 then
			local signature = HttpService:JSONEncode(rows)
			local previous = boardSignatures[key]
			if not previous or previous.contents ~= signature or os.time() - previous.sentAt >= 300 then
				enqueueContents(key, signature)
				boardSignatures[key] = {contents = signature, sentAt = os.time()}
			end
		end
	end
	local expected = {}
	if LeaderboardsEnabled then
		for _, key in ipairs({"bubbles_alltime", "bubbles_weekly", "hatches_alltime", "hatches_weekly"}) do table.insert(expected, key) end
	end
	if CompetitiveLeaderboardEnabled then
		table.insert(expected, "competitive_bubbles_current")
		table.insert(expected, "competitive_hatches_current")
	end
	for _, key in ipairs(expected) do
		if not availability[key] then
			local board = key:gsub("_[^_]+$", "")
			availability[key] = {
				rows = 0, source = "not_observed",
				reason = boardModels[board] and "period_not_loaded" or "board_not_replicated",
			}
		end
		if availability[key].rows == 0 then table.insert(missing, key) end
	end
	local notice = table.concat(missing, ", ")
	boardMissing = #missing > 0
	local context = getBoardContext()
	if boardNotice == nil or boardNotice ~= notice then
		if notice ~= "" then
			if not next(boardModels) and not next(boardPanels) then
				log("No leaderboard models are replicated in place " .. tostring(context.placeId)
					.. (context.tradingHangout and " (trading hangout)" or "")
					.. ". Waiting for public boards; no rankings were fabricated. See leaderboard_sources.json for place and loading details.")
			else
				log("Waiting for leaderboard periods: " .. notice .. ". Retrying loading checks every 15 seconds; loaded rows are exported independently.")
			end
		end
		boardNotice = notice
	end
	local diagnostic = {boards = availability, missing = missing, context = context, models = table.clone(boardModels)}
	local stableBoards = {}
	for key, info in pairs(availability) do
		stableBoards[key] = table.clone(info)
		stableBoards[key].lastAttachedAtUTC = nil
	end
	local signature = HttpService:JSONEncode({boards = stableBoards, missing = missing, context = context, models = diagnostic.models})
	if boardSourceSignature ~= signature then
		diagnostic.capturedAtUTC = os.date("!%Y-%m-%dT%H:%M:%SZ")
		enqueue("leaderboard_sources", diagnostic)
		boardSourceSignature = signature
	end
end
local function observeBoards()
	local connections = {}
	local revision = boardRevision
	local function added(node)
		if not runtime.Running then return end
		rememberBoardPanel(node, true)
	end
	table.insert(connections, workspace.DescendantAdded:Connect(added))
	table.insert(connections, workspace.DescendantRemoving:Connect(added))
	table.insert(connections, LocalPlayer.DescendantAdded:Connect(added))
	table.insert(connections, LocalPlayer.DescendantRemoving:Connect(added))
	task.spawn(function()
		while runtime.Running do
			task.wait(5)
			local current = runtime.Stages.leaderboards
			if runtime.Running and (revision ~= boardRevision or boardMissing and os.clock() - boardLastPoll >= 15)
				and (not current or current.State ~= "running") then
				revision = boardRevision
				stage("leaderboards", collectBoards, 90)
			end
		end
		for _, connection in ipairs(connections) do connection:Disconnect() end
		table.clear(boardPanels)
	end)
end

local EnchantNames = {b = "Bubbles", m = "Money", w = "Worlds", f = "Fusion", a = "Mega"}
local function decodePets(pets)
	local decoded = {}
	for _, pet in ipairs(pets) do
		assert(type(pet) == "table", "BGS: inventory contains an invalid pet record.")
		local enchantType = pet[10]
		if enchantType == nil or enchantType == false then
			enchantType = "Normal"
		else
			enchantType = EnchantNames[enchantType] or enchantType
		end
		local record = {
			id = pet[1], name = pet[2], nickname = pet[3],
			level = pet[4], xp = pet[5], equipped = pet[6] == true,
			locked = pet[7] == true, shiny = pet[8] == true,
			enchantLevel = pet[9], enchantType = enchantType, knownDupe = pet[11] == true,
		}
		local extra = {}
		for key, value in pairs(pet) do
			if type(key) ~= "number" or key < 1 or key > 11 then extra[tostring(key)] = value end
		end
		if next(extra) then record.extraFields = extra end
		table.insert(decoded, record)
	end
	return decoded
end
local function decodeHats(hats)
	local decoded = {}
	for _, hat in ipairs(hats or {}) do
		assert(type(hat) == "table", "BGS: inventory contains an invalid hat record.")
		local record = {id = hat[1], name = hat[2], equippedPetId = hat[3]}
		local extra = {}
		for key, value in pairs(hat) do
			if type(key) ~= "number" or key < 1 or key > 3 then extra[tostring(key)] = value end
		end
		if next(extra) then record.extraFields = extra end
		table.insert(decoded, record)
	end
	return decoded
end
local function inventoryRecord(player, inventory)
	local record = {
		username = player.Name, userid = player.UserId, displayname = player.DisplayName,
		pets = decodePets(inventory.Pets), hats = decodeHats(inventory.Hats),
	}
	local tags = {}
	for _, key in ipairs({"ChatTags", "Title", "EquippedTitle", "PlayerTag"}) do
		local value = player:GetAttribute(key)
		if value ~= nil then tags[key] = value end
	end
	if next(tags) then record.tags = tags end
	return record
end
local function timestamp()
	local ok, value = pcall(workspace.GetServerTimeNow, workspace)
	return ok and type(value) == "number" and value or os.time()
end
local scanQueues = {trade = {}, public = {}, normal = {}}
local scanPending, inventoryPrivate, privacyGeneration, inventoryScanned = {}, {}, {}, {}
local requestBusy = false
local nextInventoryAt, nextTradeAt = 0, 0
local tradeRosterPoll, tradeRosterNextAt = nil, math.huge
local function pendingScans()
	return #scanQueues.trade + #scanQueues.public + #scanQueues.normal
end
local function fetchInventory(player)
	if not runtime.Running then return false, "Exporter stopped." end
	if player.Parent ~= Players then return false, "Player left before inventory capture." end
	if inventoryPrivate[player] then return true, nil, nil, "Inventory is private." end
	local remote = ReplicatedStorage:FindFirstChild("NetworkRemoteFunction")
	if not remote or not remote:IsA("RemoteFunction") then return false, "Inventory remote is unavailable." end
	local generation = privacyGeneration[player]
	local startedAt = timestamp()
	local ok, inventory, reason = bounded(12, remote.InvokeServer, remote, "TradeViewInventory", player)
	if not ok then return false, redact(inventory), startedAt end
	if inventoryPrivate[player] or privacyGeneration[player] ~= generation or player.Parent ~= Players then
		return true, nil, startedAt, "Inventory privacy changed or player left during capture; response discarded."
	end
	if type(inventory) ~= "table" or type(inventory.Pets) ~= "table" then
		return true, nil, startedAt, redact(reason or "Server did not grant an inventory view.")
	end
	return true, inventory, startedAt
end
local function cancelInventoryScan(player)
	local job = scanPending[player]
	if not job then return end
	for _, queue in ipairs({scanQueues.public, scanQueues.normal}) do
		local index = table.find(queue, job)
		if index then table.remove(queue, index) end
	end
	scanPending[player] = nil
end
local function queueInventory(player, reason)
	if not ScanningEnabled or not runtime.Running or player.Parent ~= Players or inventoryPrivate[player] then return end
	local priority = reason == "inventory_public"
	if inventoryScanned[player] and not priority then return end
	local existing = scanPending[player]
	if existing then
		if priority and existing.reason ~= reason then
			local index = table.find(scanQueues.normal, existing)
			if index then
				table.remove(scanQueues.normal, index)
				existing.reason = reason
				table.insert(scanQueues.public, existing)
			end
		end
		return
	end
	if #scanQueues.public + #scanQueues.normal >= 256 then
		warnError("Inventory scan queue is full; " .. player.Name .. " will wait for the next scheduled pass.")
		return
	end
	local job = {player = player, reason = reason}
	scanPending[player] = job
	table.insert(priority and scanQueues.public or scanQueues.normal, job)
end
local function updateInventoryPrivacy(player, private, baseline)
	if typeof(player) ~= "Instance" or not player:IsA("Player") or player.Parent ~= Players or type(private) ~= "boolean" then return end
	local previous = inventoryPrivate[player]
	if previous == private then return end
	inventoryPrivate[player] = private
	privacyGeneration[player] = (privacyGeneration[player] or 0) + 1
	if private then
		cancelInventoryScan(player)
		inventoryScanned[player] = nil
	elseif not baseline or previous == true then
		inventoryScanned[player] = nil
		queueInventory(player, "inventory_public")
	end
end
local function collectInventories()
	local players = Players:GetPlayers()
	table.sort(players, function(a, b) return a.UserId < b.UserId end)
	for _, player in ipairs(players) do queueInventory(player, "scheduled") end
end
local function startInventoryQueue()
	task.spawn(function()
		while runtime.Running do
			local now = os.clock()
			local job
			if not requestBusy and now >= nextTradeAt then
				if #scanQueues.trade > 0 and #runtime.Queue < 96 then
					job = table.remove(scanQueues.trade, 1)
				elseif #scanQueues.trade == 0 and tradeRosterPoll and now >= tradeRosterNextAt then
					tradeRosterNextAt = now + 30
					job = {capture = tradeRosterPoll}
				elseif #scanQueues.trade == 0 and #runtime.Queue < 64 then
					if #scanQueues.public > 0 then
						job = table.remove(scanQueues.public, 1)
					elseif now >= nextInventoryAt then
						job = table.remove(scanQueues.normal, 1)
					end
				end
			end
			if job then
				requestBusy = true
				if not job.capture then scanPending[job.player] = nil end
				local ok, err = pcall(function()
					if job.capture then
						job.capture()
						return
					end
					local player = job.player
					if player.Parent ~= Players or inventoryPrivate[player] then return end
					local success, inventory, startedAt, reason = fetchInventory(player)
					if success and inventory then
						local record = inventoryRecord(player, inventory)
						record.reason = job.reason
						record.snapshotStartedAt = startedAt
						record.snapshotCapturedAt = timestamp()
						enqueue("inventory_" .. tostring(player.UserId), record)
						rememberExists(inventory.Pets)
						inventoryScanned[player] = true
					else
						warnError(("Inventory unavailable for %s: %s."):format(player.Name, tostring(reason or inventory)))
					end
				end)
				requestBusy = false
				nextTradeAt = os.clock() + 0.25
				if not job.capture then nextInventoryAt = os.clock() + 3.5 end
				if not ok then warnError("Queued inventory capture failed: " .. tostring(err)) end
			else
				task.wait(0.05)
			end
		end
		table.clear(scanQueues.trade)
		table.clear(scanQueues.public)
		table.clear(scanQueues.normal)
		table.clear(scanPending)
		table.clear(inventoryScanned)
	end)
end
local function observeInventoryAccess()
	local connections = {}
	table.insert(connections, Players.PlayerAdded:Connect(function(player)
		queueInventory(player, "player_joined")
	end))
	table.insert(connections, Players.PlayerRemoving:Connect(function(player)
		cancelInventoryScan(player)
		inventoryPrivate[player], privacyGeneration[player] = nil, nil
		inventoryScanned[player] = nil
	end))
	local remote = requiredChild(ReplicatedStorage, "NetworkRemoteEvent")
	assert(remote:IsA("RemoteEvent"), "BGS: inventory privacy event remote is unavailable.")
	table.insert(connections, remote.OnClientEvent:Connect(function(event, player, private)
		if not runtime.Running or event ~= "InventoryPrivacyChanged" then return end
		updateInventoryPrivacy(player, private, false)
	end))
	task.spawn(function()
		while runtime.Running do task.wait(1) end
		for _, connection in ipairs(connections) do connection:Disconnect() end
		table.clear(inventoryPrivate)
		table.clear(privacyGeneration)
	end)
end
local function collectExists()
	if not next(existsPending) then return end
	local modules = requiredChild(requiredChild(ReplicatedStorage, "Assets"), "Modules")
	local gate = safeRequire(requiredChild(modules, "IsExistsUiEnabled"))
	if type(gate) ~= "function" or gate() ~= true then
		warnError("The game has not enabled exists viewing; no count requests were made.")
		return
	end
	local remote = requiredChild(ReplicatedStorage, "NetworkRemoteFunction")
	assert(remote:IsA("RemoteFunction"), "BGS: missing inventory/exists remote.")
	local keys, rows, completed = {}, {}, {}
	for key in pairs(existsPending) do table.insert(keys, key) end
	table.sort(keys, function(a, b)
		local left, right = existsAttempted[a] or -1, existsAttempted[b] or -1
		return left < right or left == right and a < b
	end)
	for i = 1, math.min(#keys, 50) do
		while runtime.Running and (requestBusy or pendingScans() > 0 or os.clock() < nextTradeAt
			or tradeRosterPoll and os.clock() >= tradeRosterNextAt or #runtime.Queue >= 64) do
			task.wait(0.1)
		end
		if not runtime.Running then break end
		if gate() ~= true then
			warnError("Exists viewing was disabled; remaining queries are postponed.")
			break
		end
		local key = keys[i]
		local pet = existsPending[key]
		existsAttempted[key] = os.clock()
		requestBusy = true
		local ran, ok, counts = pcall(bounded, 10, remote.InvokeServer, remote, "GetPetExistsCount", pet.name, pet.shiny)
		requestBusy = false
		nextTradeAt = os.clock() + 0.25
		if not ran then ok, counts = false, ok end
		if not ok then
			warnError("Exists request failed; remaining queries are postponed.")
			break
		end
		if type(counts) == "table" and type(counts.count) == "string" and counts.count:match("^%d+$") then
			local row = table.clone(counts)
			row.name, row.shiny = pet.name, pet.shiny
			table.insert(rows, row)
			table.insert(completed, key)
		else
			warnError("Exists count unavailable for " .. pet.name .. "; no value was invented.")
		end
		task.wait(1.5)
	end
	if #rows > 0 then
		while runtime.Running and #runtime.Queue >= 64 do task.wait(0.25) end
		if not runtime.Running then return end
		enqueue("pet_exists", rows)
		for _, key in ipairs(completed) do
			existsPending[key], existsDone[key], existsAttempted[key] = nil, true, nil
		end
	end
end
local function tradingValue(value)
	if value == nil then return false end
	if type(value) == "boolean" then return value end
	if type(value) == "string" then
		local normalized = value:lower():gsub("[%s_%-]", "")
		if normalized == "trading" or normalized == "true" then return true end
		if normalized == "idle" or normalized == "nottrading" or normalized == "false" or normalized == "none" then return false end
	end
	return nil
end
local function observeTrades()
	local watched, sequence = {}, 0
	local configured = type(Settings.tradeAttribute) == "string" and Settings.tradeAttribute ~= "" and Settings.tradeAttribute or nil
	local ownParticipants, ownTradeId
	local failures, lastRosterNotice = 0, nil
	local function isPlayer(player)
		return typeof(player) == "Instance" and player:IsA("Player") and player.Parent == Players
	end
	local function queueTrade(player, state, event)
		if #scanQueues.trade >= 256 or state.pending >= 32 then
			warnError("Trade snapshot queue is full for " .. player.Name .. "; transition " .. event.sequence .. " could not be captured.")
			return
		end
		state.pending = state.pending + 1
		table.insert(scanQueues.trade, {
			player = player,
			capture = function()
				local captured, err = pcall(function()
					local ok, inventory, startedAt, reason = fetchInventory(player)
					local record
					if ok and inventory then
						record = inventoryRecord(player, inventory)
						rememberExists(inventory.Pets)
					else
						record = {username = player.Name, userid = player.UserId, displayname = player.DisplayName}
						record.inventoryError = redact(reason or inventory or "Inventory unavailable.")
					end
					record.type = event.type
					record.observedAt = event.observedAt
					record.tradeAttribute = event.attribute
					record.tradeStateSource = event.source
					record.tradeId = event.tradeId
					record.pollIntervalSeconds = event.source == "TradeGetPlayers" and 5 or nil
					record.previousObservedAt = event.previousObservedAt
					record.snapshotStartedAt = startedAt
					record.snapshotCapturedAt = timestamp()
					record.inventoryAvailable = ok and inventory ~= nil
					record.tradeStateChangedDuringCapture = state.generation ~= event.generation or player.Parent ~= Players
					enqueue(event.type .. "_" .. player.UserId, record, true)
				end)
				if not captured then warnError("Trade snapshot failed for " .. player.Name .. ": " .. tostring(err)) end
				state.pending = state.pending - 1
			end,
		})
	end
	local function transition(player, value, source, explicit, tradeId)
		local state = watched[player]
		if not state or not isPlayer(player) then return end
		local observedAt, previousObservedAt = timestamp(), state.lastObservedAt
		state.lastObservedAt = observedAt
		if state.trading == nil and not explicit then
			state.trading = value
			return
		end
		if value == state.trading then return end
		state.trading = value
		state.generation = state.generation + 1
		sequence = sequence + 1
		queueTrade(player, state, {
			type = value and "trade_started" or "trade_ended", attribute = source == "attribute" and configured or nil,
			source = source, tradeId = tradeId, observedAt = observedAt, previousObservedAt = previousObservedAt,
			generation = state.generation, sequence = sequence,
		})
	end
	local function watch(player)
		if watched[player] or player.Parent ~= Players then return end
		local state = {generation = 0, pending = 0}
		watched[player] = state
		if configured then
			local value = player:GetAttribute(configured)
			if value ~= nil then state.trading = tradingValue(value) end
			state.connection = player.AttributeChanged:Connect(function(name)
				if not runtime.Running or name ~= configured or player.Parent ~= Players then return end
				local current = player:GetAttribute(name)
				local trading
				if current ~= nil then trading = tradingValue(current) end
				if trading == nil then
					warnError("Configured trade attribute " .. name .. " is absent or unsupported on " .. player.Name .. ".")
					return
				end
				if not ownParticipants or not table.find(ownParticipants, player) then
					transition(player, trading, "attribute", false)
				end
			end)
		end
	end
	local added = Players.PlayerAdded:Connect(watch)
	local removing = Players.PlayerRemoving:Connect(function(player)
		local state = watched[player]
		if state then
			state.generation = state.generation + 1
			if state.connection then state.connection:Disconnect() end
			watched[player] = nil
		end
	end)
	for _, player in ipairs(Players:GetPlayers()) do watch(player) end
	local events = ReplicatedStorage:FindFirstChild("NetworkRemoteEvent")
	local eventConnection
	if events and events:IsA("RemoteEvent") then
		eventConnection = events.OnClientEvent:Connect(function(event, data)
			if not runtime.Running then return end
			if event == "TradeOpened" and type(data) == "table" and isPlayer(data.Player0) and isPlayer(data.Player1)
				and data.Player0 ~= data.Player1 and (data.Player0 == LocalPlayer or data.Player1 == LocalPlayer) then
				ownParticipants = {data.Player0, data.Player1}
				ownTradeId = data.Id
				for _, player in ipairs(ownParticipants) do
					watch(player)
					transition(player, true, event, true, ownTradeId)
				end
			elseif (event == "TradeCompleted" or event == "TradeCancelled") and ownParticipants then
				for _, player in ipairs(ownParticipants) do transition(player, false, event, true, ownTradeId) end
				ownParticipants, ownTradeId = nil, nil
				if not configured then tradeRosterNextAt = os.clock() + 5 end
			end
		end)
	else
		warnError("Trade event remote is absent. Own-trade start/end events cannot be observed.")
	end
	if not configured then
		tradeRosterNextAt = os.clock()
		tradeRosterPoll = function()
			local assets = ReplicatedStorage:FindFirstChild("Assets")
			local modules = assets and assets:FindFirstChild("Modules")
			local tradeModule = modules and modules:FindFirstChild("IsTradeEnabled")
			local gate = tradeModule and tradeModule:FindFirstChild("Enabled")
			local cooldown = LocalPlayer:GetAttribute("TradeCooldown")
			if not gate or not gate:IsA("BoolValue") or gate.Value ~= true
				or type(cooldown) == "number" and cooldown > timestamp() then
				tradeRosterNextAt = os.clock() + 15
				if lastRosterNotice ~= "unavailable" then
					log("Public trade-status polling is paused until trading is enabled and your trade cooldown ends. Own-trade events remain connected.")
					lastRosterNotice = "unavailable"
				end
				return
			end
			local remote = ReplicatedStorage:FindFirstChild("NetworkRemoteFunction")
			if not remote or not remote:IsA("RemoteFunction") then
				tradeRosterNextAt = os.clock() + 30
				if lastRosterNotice ~= "remote_missing" then warnError("Public trade-status remote is unavailable.") end
				lastRosterNotice = "remote_missing"
				return
			end
			local generations, privacyBefore = {}, {}
			for player, state in pairs(watched) do
				generations[player], privacyBefore[player] = state.generation, privacyGeneration[player]
			end
			local ok, rows, reason = bounded(10, remote.InvokeServer, remote, "TradeGetPlayers")
			if not ok or type(rows) ~= "table" then
				failures = failures + 1
				tradeRosterNextAt = os.clock() + math.min(60, 10 * failures)
				local message = redact(reason or rows or "invalid public trade-status response")
				if lastRosterNotice ~= message then warnError("Public trade-status query failed; retrying with backoff: " .. message) end
				lastRosterNotice = message
				return
			end
			failures, lastRosterNotice = 0, nil
			tradeRosterNextAt = os.clock() + 5
			for _, entry in ipairs(rows) do
				if type(entry) == "table" and isPlayer(entry[1]) then
					local player, status = entry[1], entry[2]
					watch(player)
					if privacyGeneration[player] == privacyBefore[player] then updateInventoryPrivacy(player, entry[4], true) end
					local value
					if status == "Trading" then value = true
					elseif status == "Trade" or status == "Disabled" or status == "Cooldown" then value = false end
					if value ~= nil and watched[player].generation == generations[player]
						and (not ownParticipants or not table.find(ownParticipants, player)) then
						transition(player, value, "TradeGetPlayers", false)
					end
				end
			end
		end
	end
	task.spawn(function()
		while runtime.Running do task.wait(1) end
		tradeRosterPoll, tradeRosterNextAt = nil, math.huge
		added:Disconnect()
		removing:Disconnect()
		if eventConnection then eventConnection:Disconnect() end
		for _, state in pairs(watched) do if state.connection then state.connection:Disconnect() end end
	end)
	log(configured and ("Watching configured trade attribute " .. configured .. " and own-trade events.")
		or "Watching own-trade events and polling public trade statuses every 5 seconds when permitted. First observations are baselines; trades between polls can be missed.")
end
local function snapshotStages()
	if ScanningEnabled then
		stage("inventories", collectInventories, 30)
	end
	if game.PlaceId == BGS_PLACE_ID or game.PlaceId == TRADE_HANGOUT_PLACE_ID then
		if LeaderboardsEnabled or CompetitiveLeaderboardEnabled then
			stage("leaderboards", collectBoards, 90)
		end
	end
end
local function collectImageService()
	local Assets = requiredChild(ReplicatedStorage, "Assets")
	local Modules = requiredChild(Assets, "Modules")
	local ImageScript = requiredChild(Modules, "ImageService")
	local service = safeRequire(ImageScript)
	local success, source = pcall(function()
		return ImageScript.Source
	end)
	local images = {}
	local sources = {
		requiredChild(requiredChild(Modules, "ItemDataService"), "PetModule"),
		requiredChild(requiredChild(Modules, "ItemDataService"), "HatModule"),
		requiredChild(requiredChild(Modules, "ItemDataService"), "EggModule"),
		requiredChild(requiredChild(Modules, "ItemDataService"), "CrateModule"),
	}
	for _, moduleScript in ipairs(sources) do
		local ok, module = pcall(require, moduleScript)
		if ok and type(module) == "table" then
			for name, value in pairs(module) do
				if type(name) == "string" and type(value) == "table" then
					local called, image = pcall(service, name)
					if called and image ~= nil then images[name] = image end
				end
			end
		end
	end
	enqueue("imageservice", {
		Module = ImageScript:GetFullName(),
		Source = success and type(source) == "string" and source ~= "" and source or nil,
		SourceAvailable = success and type(source) == "string" and source ~= "",
		Images = images,
	})
end
if ScanningEnabled or TradeScanEnabled then
	stage("inventory_access_observer", observeInventoryAccess, 30)
	startInventoryQueue()
end
if TradeScanEnabled then stage("trade_observer", observeTrades, 30) end
if (game.PlaceId == BGS_PLACE_ID or game.PlaceId == TRADE_HANGOUT_PLACE_ID) and (LeaderboardsEnabled or CompetitiveLeaderboardEnabled) then
	stage("leaderboard_observer", observeBoards, 30)
end
snapshotStages()
if ImageServiceEnabled then
	stage("imageservice", collectImageService, 30)
end
task.spawn(function()
	local day = math.floor(os.time() / 86400)
	while runtime.Running and routingActive and not tradeHangoutReady do task.wait(1) end
	if not runtime.Running then return end
	if (game.PlaceId == BGS_PLACE_ID or game.PlaceId == TRADE_HANGOUT_PLACE_ID) and LeaderboardsEnabled then
		while runtime.Running and boardMissing do
			local ok, err = pcall(collectBoards)
			if not ok then warnError("Leaderboard collection failed: " .. tostring(err)) end
			if boardMissing then task.wait(15) end
		end
	end
	if enabled(Settings.bounties) then stage("bounties", collectBounties, 120) end
	if enabled(Settings.pets) or enabled(Settings.eggs) then stage("catalog", collectCatalog, 180) end
	if ExistsEnabled then
		task.spawn(function()
			while runtime.Running and tradeHangoutReady do
				local ok, err = pcall(collectExists)
				if not ok then
					warnError("Exists scan failed: " .. tostring(err))
				end
				task.wait(30)
			end
		end)
	end
	task.wait(300)
	while runtime.Running do
		snapshotStages()
		local nextDay = math.floor(os.time() / 86400)
		if nextDay ~= day then
			day = nextDay
			if enabled(Settings.bounties) then stage("bounties", collectBounties, 120) end
		end
		task.wait(300)
	end
end)
return runtime
