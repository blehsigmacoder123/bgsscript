local function parseOriginalStatKeys(source)
local result = {}
if type(source) ~= "string" then
return result
end
local currentName
local readingBuffs = false
for rawLine in source:gmatch("[^\r\n]+") do
local line = rawLine:gsub("%-%-.*$", "")
if readingBuffs then
if line:match("^%s*}%s*,?%s*$") then
readingBuffs = false
else
local key =
line:match('^%s*%["(.-)"%]%s*=')
or line:match("^%s*([%a_][%w_]*)%s*=")
if key then
table.insert(result[currentName], key)
end
end
else
local quotedName = line:match('^%s*%["(.-)"%]%s*=%s*{%s*,?%s*$')
local plainName = line:match("^%s*([%a_][%w_]*)%s*=%s*{%s*,?%s*$")
if quotedName then
currentName = quotedName
elseif plainName and plainName ~= "Buffs" then
currentName = plainName
end
if currentName and line:match("^%s*Buffs%s*=%s*{%s*,?%s*$") then
result[currentName] = {}
readingBuffs = true
end
end
end
return result
end

local function cleanStableImageUrl(value)
if type(value) ~= "string" then
return nil
end
value = value:gsub("\\u0026", "&")
value = value:match("^%s*(.-)%s*$") or value
if not value:match("^https://tr%.rbxcdn%.com/") then
return nil
end
return value:match("^[^?]+")
end
local function extractAssetId(value)
if type(value) == "number" then
return tostring(math.floor(value))

if type(value) ~= "string" then
return nil

or value:match("[?&]id=(%d+)")
or value:match("rbxthumb://.-id=(%d+)")
or value:match("^%s*(%d+)%s*$")
end

local resolvedImagesByAssetId = {}
local function fetchThumbnailBatch(batch)
if #batch == 0 then
return
end
local url =
"https://thumbnails.roblox.com/v1/assets?assetIds="
.. table.concat(batch, ",")
.. "&size="
.. THUMBNAIL_SIZE
.. "&format=Png&isCircular=false"
local response = request({
Url = url,
Method = "GET",
Headers = {
["Accept"] = "application/json",
["User-Agent"] = "Mozilla/5.0",
},
})
if responseStatus(response) ~= 200 then
return
end
local body = responseBody(response)
local success, decoded = pcall(HttpService.JSONDecode, HttpService, body or "")
if not success or type(decoded) ~= "table" or type(decoded.data) ~= "table" then
return
end
for _, item in ipairs(decoded.data) do
local assetId = item.targetId and string.format("%.0f", item.targetId)
local stableUrl = cleanStableImageUrl(item.imageUrl)
if assetId and item.state == "Completed" and stableUrl then
resolvedImagesByAssetId[assetId] = stableUrl
end
end
end
local unresolved = table.clone(assetIds)
for attempt = 1, THUMBNAIL_RETRIES do
for startIndex = 1, #unresolved, THUMBNAIL_BATCH_SIZE do
local batch = {}
local endIndex = math.min(startIndex + THUMBNAIL_BATCH_SIZE - 1, #unresolved)
for index = startIndex, endIndex do
table.insert(batch, unresolved[index])
end
fetchThumbnailBatch(batch)
task.wait(0.1)
end
local nextUnresolved = {}
for _, assetId in ipairs(unresolved) do
if not resolvedImagesByAssetId[assetId] then
table.insert(nextUnresolved, assetId)
end
end
unresolved = nextUnresolved
if #unresolved == 0 then
break
end
if attempt < THUMBNAIL_RETRIES then
task.wait(1.5)
end
end

Url = url,
Method = method,
Redirect = false,

["User-Agent"] = "Mozilla/5.0",
},
})

response.Url,
response.URL,
response.ResponseUrl,

response.FinalUrl,
response.FinalURL,
getHeader(response.Headers or response.headers, "location"),
}
for _, candidate in ipairs(candidates) do

if stableUrl then

end
end
local body = responseBody(response)
if type(body) == "string" then
local found = body:match("https://tr%.rbxcdn%.com/[^\"'%s<>]+")
local stableUrl = cleanStableImageUrl(found)

return stableUrl
end
end
end
end
return nil
end

local function buildPermutation(seed, count, salt)
local permutation = {}
for index = 1, count do
permutation[index] = index
end
local random = Random.new(seed * 131 + salt)
for index = count, 2, -1 do
local swapIndex = random:NextInteger(1, index)
permutation[index], permutation[swapIndex] = permutation[swapIndex], permutation[index]
end
return permutation
end
local function getRotatingIndex(seed, count, salt)
local cycle = math.floor(seed / count)
local current = buildPermutation(cycle, count, salt)
if count > 2 then
local previous = buildPermutation(cycle - 1, count, salt)
if current[1] == previous[count] then
current[1], current[2] = current[2], current[1]
end
end
return current[seed % count + 1]
end

local function uploadJsonAttachment(webhookUrl, attachmentName, contents)
local boundary = "----------------BGSI" .. HttpService:GenerateGUID(false):gsub("-", "")
local payloadJson = HttpService:JSONEncode({
content = ("BGSI data export: %d pets, %d hats, %d eggs, %d crates, %d bounties."):format(
#output.Pets,
#output.Hats,
#output.Eggs,
#output.Crates,
#output.SecretBounties
),
})
local body = table.concat({
"--" .. boundary,
'Content-Disposition: form-data; name="payload_json"',
"Content-Type: application/json",
"",
payloadJson,
"--" .. boundary,
('Content-Disposition: form-data; name="files[0]"; filename="%s"'):format(attachmentName),
"Content-Type: application/json",
"",
contents,
"--" .. boundary .. "--",
"",
}, "\r\n")
return request({
Url = webhookUrl,
Method = "POST",
Headers = {
["Content-Type"] = "multipart/form-data; boundary=" .. boundary,
["User-Agent"] = "Mozilla/5.0",
},
Body = body,
})
end
local response = uploadJsonAttachment(WEBHOOK_URL, filename, json)
local status = responseStatus(response)
if status and status >= 200 and status < 300 then
print(("Export uploaded successfully as %s (%d bytes)."):format(filename, #json))
else
error(("Webhook upload failed. Status: %s Body: %s"):format(
tostring(status),
tostring(responseBody(response))
))
end
