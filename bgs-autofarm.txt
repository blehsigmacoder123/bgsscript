repeat task.wait() until game:IsLoaded()

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")
local VirtualUser = game:GetService("VirtualUser")

local player = Players.LocalPlayer
local remoteEvent = ReplicatedStorage:WaitForChild("NetworkRemoteEvent")
local remoteFunction = ReplicatedStorage:WaitForChild("NetworkRemoteFunction")
local leaderstats = player:WaitForChild("leaderstats")
local eggsOpened = leaderstats:WaitForChild("Eggs Opened")
local gems = leaderstats:WaitForChild("Gems")

local settings = {
    hatchTarget = 3000,
    requestDelay = 0.45,
    hatchRequestDelay = 0.12,
    hatchDelay = 0.12,
    autoDeleteRefresh = 1.5
}

local islandRoute = {
    "The Floating Island",
    "Space",
    "The Skylands",
    "The Twilight",
    "The Void",
    "Zen",
    "XP Island"
}

local running = true

local function fire(name, ...)
    local arguments = table.pack(...)
    local ok = pcall(function()
        remoteEvent:FireServer(name, table.unpack(arguments, 1, arguments.n))
    end)
    task.wait(settings.requestDelay)
    return ok
end

local function purchaseEgg(name)
    local ok = pcall(function()
        remoteEvent:FireServer("PurchaseEgg", name)
    end)
    task.wait(settings.hatchRequestDelay)
    return ok
end

local function invoke(name, ...)
    local arguments = table.pack(...)
    local ok, result = pcall(function()
        return remoteFunction:InvokeServer(name, table.unpack(arguments, 1, arguments.n))
    end)
    task.wait(settings.requestDelay)
    if ok then
        return result
    end
end

local function getData()
    local data = invoke("GetPlayerData")
    if type(data) == "table" then
        return data
    end
end

local function getRoot()
    local character = player.Character or player.CharacterAdded:Wait()
    return character:WaitForChild("HumanoidRootPart")
end

local function hasCheckpoint(data, name)
    local checkpoints = data and data[11]
    if type(checkpoints) ~= "table" then
        return false
    end
    return table.find(checkpoints, name) ~= nil
end

local function travelToIsland(name, attempt)
    local overworld = workspace.FloatingIslands:FindFirstChild("Overworld")
    local island = overworld and overworld:FindFirstChild(name)
    local destination = island and island:FindFirstChild("TeleportPoint")
    if not destination or not destination:IsA("BasePart") then
        return false
    end
    local root = getRoot()
    local depth = attempt > 1 and math.min(6 + (attempt - 2) * 4, 30) or 0
    local entry = destination.CFrame * CFrame.new(0, 20, 0)
    local contact = destination.CFrame * CFrame.new(0, -depth, 0)
    local landing = destination.CFrame * CFrame.new(0, 3 - depth, 0)
    root.CFrame = entry
    task.wait(0.35)
    local tween = TweenService:Create(root, TweenInfo.new(1.25, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
        CFrame = contact
    })
    tween:Play()
    tween.Completed:Wait()
    if type(firetouchinterest) == "function" then
        pcall(firetouchinterest, root, destination, 0)
        task.wait(0.1)
        pcall(firetouchinterest, root, destination, 1)
    end
    root.CFrame = landing
    task.wait(1.5)
    return true
end

local function unlockIslands()
    for _, name in ipairs(islandRoute) do
        local attempt = 0
        while running and not hasCheckpoint(getData(), name) do
            attempt += 1
            travelToIsland(name, attempt)
            task.wait(1.25)
        end
    end
end

local tutorialTracker = {
    stage = 1,
    progress = 0,
    completed = false
}

local function updateTutorialTracker()
    local data = getData()
    local tutorial = data and data[137]
    if type(tutorial) == "table" then
        tutorialTracker.stage = math.max(1, tonumber(tutorial.Stage) or 1)
        tutorialTracker.progress = math.max(0, tonumber(tutorial.Progress) or 0)
        tutorialTracker.completed = tutorial.Completed == true or tutorialTracker.stage > 6
    end
    return tutorialTracker
end

local function moveNearEgg(name)
    local egg = workspace.Eggs:FindFirstChild(name)
    local hotkey = egg and egg:FindFirstChild("Hotkey")
    if not hotkey or not hotkey:IsA("BasePart") then
        return false
    end
    local root = getRoot()
    root.CFrame = hotkey.CFrame * CFrame.new(0, 3, 5)
    task.wait(0.5)
    return (root.Position - hotkey.Position).Magnitude <= 12
end

local tutorialActions = {
    [1] = function()
        for _ = 1, 5 do
            fire("BlowBubble")
        end
        fire("SellBubble", "Sell")
    end,
    [2] = function()
        fire("BuyShopItem", "EarthFlavors", "Flavors", "Blueberry")
    end,
    [3] = function()
        for _ = 1, 5 do
            fire("BlowBubble")
        end
        fire("SellBubble", "Sell")
    end,
    [4] = function()
        if moveNearEgg("Common Egg") then
            fire("PurchaseEgg", "Common Egg")
            task.wait(settings.hatchDelay)
        end
    end,
    [5] = function()
        fire("BuyShopItem", "EarthGum", "Gum", "Stretchy Gum")
    end,
    [6] = function()
        if moveNearEgg("Spotted Egg") then
            fire("PurchaseEgg", "Spotted Egg")
            task.wait(settings.hatchDelay)
        end
    end
}

local function completeTutorial()
    while running do
        local state = updateTutorialTracker()
        if state.completed then
            return true
        end
        local action = tutorialActions[state.stage]
        if action then
            action()
        else
            task.wait(1)
        end
        task.wait(0.75)
    end
    return false
end

local function equipBestPets()
    fire("EquipBestPets")
    task.wait(1)
end

local function useLevelPotions()
    local data = getData()
    local pets = data and data[10]
    local potions = data and data[64]
    if type(pets) ~= "table" or type(potions) ~= "table" then
        return
    end
    local petIds = {}
    for _, pet in ipairs(pets) do
        if type(pet) == "table" and pet[1] then
            table.insert(petIds, pet[1])
        end
    end
    if #petIds == 0 then
        return
    end
    local petIndex = 1
    for _, potion in ipairs(potions) do
        local potionType = type(potion) == "table" and potion[2]
        if potion[1] and type(potionType) == "string" and potionType:sub(1, 5) == "Level" then
            fire("UsePotionOnPet", potion[1], petIds[petIndex])
            petIndex = petIndex % #petIds + 1
        end
    end
end

local function enchantAllPets()
    local data = getData()
    local pets = data and data[10]
    if type(pets) ~= "table" then
        return
    end
    for _, pet in ipairs(pets) do
        if type(pet) == "table" and pet[1] and (tonumber(pet[9]) or 0) < 40 then
            fire("EnchantPet", pet[1])
        end
    end
end

local function isChestOnCooldown(name)
    local data = getData()
    local regenTimes = data and data[12]
    local regenTime = type(regenTimes) == "table" and regenTimes[name]
    return type(regenTime) == "number" and regenTime > os.time()
end

local function getChestClaimPart(name)
    local chest
    for _, world in ipairs(workspace.FloatingIslands:GetChildren()) do
        local island = world:FindFirstChild(name)
        if island then
            chest = island:FindFirstChild("Chest")
            if chest then
                break
            end
        end
    end
    local claim = chest and chest:FindFirstChild("Chest Collect " .. name)
    if not claim then
        return
    end
    if claim:IsA("BasePart") then
        return claim
    end
    if claim:IsA("Attachment") then
        return claim.Parent and claim.Parent:IsA("BasePart") and claim.Parent or nil
    end
    if claim:IsA("Model") then
        return claim.PrimaryPart or claim:FindFirstChildWhichIsA("BasePart", true)
    end
    return claim.Parent and claim.Parent:IsA("BasePart") and claim.Parent or nil
end

local function moveToChestClaim(name, attempt)
    local target = getChestClaimPart(name)
    if not target then
        return false
    end
    local root = getRoot()
    local offset = 3 - math.min((attempt - 1) * 2, 6)
    root.CFrame = target.CFrame * CFrame.new(0, offset, 0)
    task.wait(0.5)
    if type(firetouchinterest) == "function" then
        pcall(firetouchinterest, root, target, 0)
        pcall(firetouchinterest, root, target, 1)
    end
    return true
end

local function collectChest(name)
    if isChestOnCooldown(name) then
        return true
    end
    for attempt = 1, 4 do
        moveToChestClaim(name, attempt)
        fire("CollectChestReward", name)
        task.wait(0.75)
        if isChestOnCooldown(name) then
            return true
        end
    end
    return false
end

local function hatchIceShardEgg()
    local attempts = 0
    while running and gems.Value >= 250 do
        attempts += 1
        if attempts == 1 or attempts % 12 == 0 then
            moveNearEgg("Ice Shard Egg")
        end
        purchaseEgg("Ice Shard Egg")
        task.wait(settings.hatchDelay)
    end
end

local commonEggPets = { "Doggy", "Kitty", "Bunny", "Bear", "King Doggy" }

local function enableCommonEggAutoDelete()
    for _, pet in ipairs(commonEggPets) do
        pcall(function()
            remoteEvent:FireServer("AutoDelete", pet, true)
        end)
        task.wait(0.08)
    end
end

local function skipHatches()
    while running do
        local gui = player:FindFirstChildOfClass("PlayerGui")
        local screenGui = gui and gui:FindFirstChild("ScreenGui")
        if screenGui then
            for _, name in ipairs({ "PetHatch", "MultiPetHatch" }) do
                local hatch = screenGui:FindFirstChild(name)
                local skip = hatch and hatch:FindFirstChild("Skip", true)
                if skip and skip:IsA("GuiButton") and skip.Visible then
                    pcall(function()
                        skip:Activate()
                    end)
                    if type(firesignal) == "function" then
                        pcall(firesignal, skip.MouseButton1Down)
                        pcall(firesignal, skip.MouseButton1Click)
                        pcall(firesignal, skip.Activated)
                    end
                    if type(getconnections) == "function" then
                        for _, connection in ipairs(getconnections(skip.MouseButton1Down)) do
                            pcall(connection.Function)
                        end
                        for _, connection in ipairs(getconnections(skip.MouseButton1Click)) do
                            pcall(connection.Function)
                        end
                    end
                end
            end
        end
        task.wait(0.03)
    end
end

local function hatchCommonEgg()
    if not moveNearEgg("Common Egg") then
        return
    end
    task.spawn(function()
        while running and eggsOpened.Value < settings.hatchTarget do
            enableCommonEggAutoDelete()
            task.wait(settings.autoDeleteRefresh)
        end
    end)
    while running and eggsOpened.Value < settings.hatchTarget do
        purchaseEgg("Common Egg")
        task.wait(settings.hatchDelay)
    end
end

player.Idled:Connect(function()
    pcall(function()
        VirtualUser:CaptureController()
        VirtualUser:ClickButton2(Vector2.new())
    end)
end)

task.spawn(skipHatches)
completeTutorial()
unlockIslands()
collectChest("The Void")
hatchIceShardEgg()
equipBestPets()
collectChest("XP Island")
collectChest("The Floating Island")
enableCommonEggAutoDelete()
hatchCommonEgg()
running = false
